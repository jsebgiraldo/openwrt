# Bootloader configuration - R1000 base distroconfig
################################################################################
# Common CM4 settings for the Seeed reComputer Industrial R1000 carrier.
# Unlike the EKH01 base distroconfig, UART5 is NOT enabled here because
# GPIO12/13 (its pins) are claimed by the R1000 for the PCA9535 IO expander.
# UART0 (ttyAMA0 on GPIO14/15) is kept enabled for serial console.

# Restore PL011 (ttyAMA0) to GPIOs 14 & 15 on Pi4 by moving Bluetooth to the
# mini UART; this is what gives us the debug serial console on the 40-pin header.
[pi0w]
dtoverlay=disable-bt
[pi3]
dtoverlay=disable-bt
[pi4]
dtoverlay=miniuart-bt

[all]
# Disable overscan assuming the display supports displaying the full resolution
# If the text shown on the screen disappears off the edge, comment this out
disable_overscan=1

# How much memory in MB to assign to the GPU
gpu_mem_256=128
gpu_mem_512=128
gpu_mem_1024=128

# Enable UART0 (ttyAMA0, PL011) serial console on GPIO14/15.
uart_enable=1

# Disable SD activity LED by default (we may re-trigger it in userspace).
dtparam=act_led_trigger=none

# Ramoops for crash reports
dtoverlay=ramoops

# Auto-load camera overlays if present
camera_auto_detect=1
